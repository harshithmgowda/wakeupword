# Reference list of the training environment's dependencies.
#
# ./setup_env.sh IS THE AUTHORITATIVE INSTALLER -- use it, not this file.
# Two things here cannot be expressed as a flat requirements list:
#
#   1. torch must come from the cu121 index (--index-url), installed BEFORE
#      everything else so the CUDA build wins over any transitive CPU pin.
#   2. onnxruntime-gpu and onnxruntime provide the same `onnxruntime` module,
#      so only one may be installed. The GPU wheel is required: with the CPU
#      wheel, feature extraction silently runs single-threaded and the augment
#      stage takes ~7.5x longer.
#
# This file exists so you can see the pinned set at a glance and for tooling
# that expects a requirements.txt.

# --- CUDA build, from https://download.pytorch.org/whl/cu121 ---------------
torch==2.5.0
torchvision==0.20.0
torchaudio==2.5.0

# --- pins ------------------------------------------------------------------
# numpy is held below 2.0: piper-sample-generator, audiomentations 0.33 and
# speechbrain 0.5.14 all predate the numpy 2 ABI break.
numpy<2
# webrtcvad and pronouncing both import pkg_resources, removed in setuptools 81.
setuptools<81

mutagen==1.47.0
